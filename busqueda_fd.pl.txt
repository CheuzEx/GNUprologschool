% ============================================================
% busqueda_fd.pl -- Estrategia CLP(FD)
% Consultar DESPUES de horarios.pl, valida.pl y busqueda_bt.pl.
%
% Aporta:
%   - buscar(clpfd, Opc, Pares, Sol)
%   - preparar_indices/0, construir_vars/2, imponer_restricciones/3
%   - etiquetar_fd/2, reconstruir/3
%
% Uso tipico:
%   resolver(clpfd, [], R).
%   resolver(clpfd, [heuristica(ffc), simetria(si), limite(500000)], R).
%   resolver(clpfd, [heuristica(ff), optimizar, limite(500000)], R).
%
% Notas de integracion:
%   - Este es el UNICO archivo que importa library(clpfd).
%     La directiva DEBE ir al inicio, antes de cualquier clausula.
%   - Reutiliza: opcion/4, excluyentes_par/4, reiniciar_contadores/1,
%     g_read(limite,L), g_assign/2, g_inc/1 (de busqueda_bt.pl).
%   - costo_total/2 (o costo_blandas/2) debe existir en el sistema
%     para que funcione la opcion optimizar.
% ============================================================

:- use_module(library(clpfd)).

:- dynamic(fd_indice/3).      % fd_indice(aula|dia, Indice, Codigo)

% ------------------------------------------------------------
% Punto de entrada (misma firma que buscar/4 de busqueda_bt.pl)
% ------------------------------------------------------------
buscar(clpfd, Opc, Pares, Sol) :-
    opcion(heuristica, Opc, original, H),
    opcion(simetria,    Opc, no,       Sim),
    preparar_indices,
    construir_vars(Pares, Vars),
    imponer_restricciones(Pares, Vars, Sim),
    (   member(optimizar, Opc)
    ->  buscar_optimo(Vars, Pares, H, Opc, Sol)
    ;   etiquetar_fd(Vars, H, Opc),
        reconstruir(Pares, Vars, Sol)
    ).
% ------------------------------------------------------------
% Indices aula/dia <-> entero consecutivo
% ------------------------------------------------------------
preparar_indices :-
    retractall(fd_indice(_,_,_)),
    findall(A, aula(A,_,_), Aulas),
    numerar(Aulas, 0, aula),
    findall(D, dia(D), Dias),
    numerar(Dias, 0, dia).

numerar([], _, _).
numerar([X|Xs], N, Tipo) :-
    assertz(fd_indice(Tipo, N, X)),
    N1 is N + 1,
    numerar(Xs, N1, Tipo).

indice(Tipo, Valor, Idx) :- fd_indice(Tipo, Idx, Valor).
valor(Tipo, Idx, Valor)  :- fd_indice(Tipo, Idx, Valor).

% ------------------------------------------------------------
% Variables FD: una terna (A, D, F) por sesion
% Se usan 3 variables FD por sesion y fd_relation/2 para restringir
% la combinacion a las tuplas validas del dominio precalculado.
% ------------------------------------------------------------
construir_vars([], []).
construir_vars([S-Dom|Ps], [fv(S, A, D, F)|Vs]) :-
    tuplas_indices(Dom, Tuplas),
    aulas_de(Tuplas, As),
    dias_de(Tuplas, Ds),
    franjas_de(Tuplas, Fs),
    fd_domain(A, As),
    fd_domain(D, Ds),
    fd_domain(F, Fs),
    fd_relation(Tuplas, [A, D, F]),
    construir_vars(Ps, Vs).

tuplas_indices(Dom, Tuplas) :-
    findall([IA, ID, F],
            ( member(v(A, D, F), Dom),
              indice(aula, A, IA),
              indice(dia,  D, ID)
            ),
            Tuplas).

aulas_de(Tuplas, L) :-
    findall(IA, member([IA|_], Tuplas), L0), sort(L0, L).
dias_de(Tuplas, L) :-
    findall(ID, member([_,ID|_], Tuplas), L0), sort(L0, L).
franjas_de(Tuplas, L) :-
    findall(F, member([_,_,F], Tuplas), L0), sort(L0, L).

% ------------------------------------------------------------
% Restricciones de a pares
% Se imponen inline (no se recolectan en lista) para que los
% propagadores de CLP(FD) se instalen cuanto antes.
% ------------------------------------------------------------
imponer_restricciones(Pares, Vars, Sim) :-
    pares_restriccion(Pares, Vars, Sim).

pares_restriccion([], [], _).
pares_restriccion([P1|Ps1], [V1|Vs1], Sim) :-
    pares_con_uno(P1, V1, Ps1, Vs1, Sim),
    pares_restriccion(Ps1, Vs1, Sim).

pares_con_uno(_, _, [], [], _).
pares_con_uno(P1, V1, [P2|Ps], [V2|Vs], Sim) :-
    (   poda_par(P1, V1, P2, V2)
    ->  restringir_par(P1, V1, P2, V2, Sim)
    ;   true
    ),
    pares_con_uno(P1, V1, Ps, Vs, Sim).

% Solo genera restriccion si el par puede realmente colisionar
poda_par(S1-_, fv(S1,_,_,_), S2-_, fv(S2,_,_,_)) :-
    S1 = sesion(_, C1, G1, _, P1, _),
    S2 = sesion(_, C2, G2, _, P2, _),
    (   P1 == P2
    ;   C1 == C2, G1 == G2
    ;   excluyentes_par(C1, G1, C2, G2)
    ).

% No solapamiento condicionado al recurso compartido
restringir_par(S1-_, fv(S1, A1, D1, F1), S2-_, fv(S2, A2, D2, F2), Sim) :-
    S1 = sesion(_, C1, G1, I1, P1, Dur1),
    S2 = sesion(_, C2, G2, I2, P2, Dur2),
    NoOverlap = ((F1 + Dur1 #=< F2) #\/ (F2 + Dur2 #=< F1)),
    % Mismo aula y mismo dia -> no solapan
    ((A1 #= A2) #/\ (D1 #= D2)) #==> NoOverlap,
    % Mismo profesor, mismo grupo o excluyentes -> no solapan
    (   (   P1 == P2
        ;   C1 == C2, G1 == G2
        ;   excluyentes_par(C1, G1, C2, G2)
        )
    ->  (D1 #= D2) #==> NoOverlap
    ;   true
    ),
    % Reduccion de simetria: sesion i < j del mismo grupo => i antes que j
    (   Sim == si, C1 == C2, G1 == G2, I1 < I2
    ->  T1 #= D1 * 1000 + F1,
        T2 #= D2 * 1000 + F2,
        T1 #=< T2
    ;   true
    ).

% ------------------------------------------------------------
% Etiquetado
%
% - Con optimizar: NO se pasa backtrack/1, porque buscar_optimo/4
%   explora el arbol completo y necesita reetiquetar. El limite se
%   respeta de forma aproximada por el reloj global (ver nota).
% - Sin optimizar: backtrack(L) acota la exploracion.
% - Si fd_labeling falla, distinguimos "limite" de "sin solucion"
%   reintentando sin backtrack. Si el reintento halla solucion,
%   era el limite; si no, es insatisfacibilidad.
% ------------------------------------------------------------
etiquetar_fd(Vars, H, _Opc) :-
    aplanar(Vars, Lista),
    etiquetas(H, Opts0),
    g_read(limite, L),
    Opts = [backtrack(L)|Opts0],
    (   fd_labeling(Lista, Opts)
    ->  true
    ;   (   fd_labeling(Lista, [ff])
        ->  throw(limite_alcanzado)
        ;   fail
        )
    ).

aplanar([], []).
aplanar([fv(_,A,D,F)|Vs], [A,D,F|Rs]) :- aplanar(Vs, Rs).

% Mapeo heuristica -> opciones de fd_labeling/2
% (ff = first fail = MRV; ffc = ff + propagacion completa)
etiquetas(original, []).
etiquetas(mrv,      [ff]).
etiquetas(ff,       [ff]).
etiquetas(ffc,      [ffc]).
etiquetas(lcv,      [ff]).       % aproximacion: ff + orden previo
etiquetas(grado,    [ff]).       % aproximacion: ff + orden previo
etiquetas(demanda,  [ff]).       % aproximacion: ff + orden previo
etiquetas(_,        []).

% ------------------------------------------------------------
% Reconstruir solucion
% ------------------------------------------------------------
reconstruir([], [], []).
reconstruir([S-_|Ps], [fv(_,A,D,F)|Vs], [S-v(Aula,Dia,Franja)|Rs]) :-
    valor(aula, A, Aula),
    valor(dia,  D, Dia),
    Franja = F,
    reconstruir(Ps, Vs, Rs).
% ------------------------------------------------------------
% Busqueda con optimizacion de restricciones blandas
%
% Estrategia: enumera todas las soluciones factibles del modelo
% FD y conserva la de menor costo (prefiere + compacta). Esto
% garantiza encontrar el optimo global.
%
buscar_optimo(Vars, Pares, H, _Opc, Sol) :-
    aplanar(Vars, Lista),
    etiquetas(H, Opts0),
    g_read(limite, L),
    Opts = [backtrack(L)|Opts0],
    g_assign(opt_mejor, none),
    g_assign(opt_costo, 1000000),
    (   fd_labeling(Lista, Opts),
        reconstruir(Pares, Vars, Sol0),
        a_horario(Sol0, H0),              % <-- convierte a asignacion/7
        costo_blandas(H0, C),             % <-- ahora sí calcula bien
        g_read(opt_costo, CActual),
        (   C < CActual
        ->  g_assign(opt_mejor, Sol0),    % <-- guarda en formato interno
            g_assign(opt_costo, C)
        ;   true
        ),
        fail
    ;   true
    ),
    g_read(opt_mejor, Sol),
    Sol \== none.
